// Exemplary Pattern: Food Emission Calculation
class CarbonCalculator {
  // Main method - simple and focused
  calculateFoodEmissions(input: FoodInput): number {
    const factor = this.getFoodEmissionFactor(input);
    const emissions = factor * input.quantity;
    return this.roundToTwoDecimals(emissions);
  }
  
  // Strategy pattern implementation
  private determineFoodFactor(input: FoodInput): number {
    switch (input.type) {
      case 'vegetables': return this.getVegetableFactor(input);
      case 'processed': return EMISSION_FACTORS.food.processed;
      case 'meat':
      case 'dairy': return this.getAnimalProductFactor(input);
      default: return 0;
    }
  }
  
  // Focused helper methods
  private getVegetableFactor(input: FoodInput): number {
    return input.isLocal 
      ? EMISSION_FACTORS.food.vegetables.local
      : EMISSION_FACTORS.food.vegetables[input.isOrganic ? 'organic' : 'standard'];
  }
}

// Safe array access with bounds checking
private safeArrayAccess<T>(array: T[], index: number): T | undefined {
  const safeIndex = index % array.length;
  if (safeIndex >= 0 && safeIndex < array.length) {
    return array[safeIndex];
  }
  return undefined;
}
